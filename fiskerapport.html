function showReports(){ const reports = JSON.parse( localStorage.getItem( "fishingReports" ) || "[]" ); let html = ""; reports.reverse().forEach(r => { html += `
${r.datum}
Plats: ${r.plats}
Resultat: Gädda ${r.gadda}, Abborre ${r.abborre}, Gös ${r.gos}
Betyg: ${r.betyg}
Prognos gädda: ${r.prognos?.pikeScore ?? "-"}
Prognos abborre: ${r.prognos?.perchScore ?? "-"}
Prognos gös: ${r.prognos?.zanderScore ?? "-"}
Lufttryck: ${r.prognos?.pressure ?? "-"}
Vind: ${r.prognos?.wind ?? "-"} m/s
Regn: ${r.prognos?.rain ?? "-"} mm
Vattentemp: ${r.prognos?.waterTemp ?? "-"} °C
Kommentar: ${r.kommentar}
`; }); document.getElementById("reports") .innerHTML = html; }
